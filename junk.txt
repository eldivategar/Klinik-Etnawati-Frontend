import axios from "axios";
import { refreshAccessToken } from "./utils";
import { useAuthStore } from "@/store/pinia/Auth";

const config = {
  baseURL: process.env.VUE_APP_API_ENDPOINT,
};

const axiosInstance = axios.create(config);

axiosInstance.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem("access");
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => {
    return Promise.reject(error);
  }
);

let isRefreshing = false;
let subscribers = [];

const onRrefreshed = (token) => {
  subscribers.map((callback) => callback(token));
};

const addSubscriber = (callback) => {
  subscribers.push(callback);
};

axiosInstance.interceptors.response.use(
  (response) => {
    return response;
  },
  async (error) => {
    const authStore = useAuthStore();
    const originalRequest = error.config;

    if (
      error.response &&
      error.response.status === 401 &&
      originalRequest.url !== "account/auth/login" &&
      error.response.data.messages[0].token_type === "access" &&
      !isRefreshing
    ) {
      if (!isRefreshing) {
        isRefreshing = true;
        originalRequest._retry = true;

        try {
          const newToken = await refreshAccessToken();
          onRrefreshed(newToken);
          isRefreshing = false;
          return axiosInstance(originalRequest);
        } catch (refreshError) {
          this.$swal.fire({
            title: "Upss!",
            text: "Sesi login Anda telah habis. Silakan login kembali.",
            icon: "warning",
            button: "OK!",
          });
          authStore.logoutWithoutRefresh();
          window.location.href = "/auth/login";
          return Promise.reject(refreshError);
        }
      } else {
        return new Promise((resolve) => {
          addSubscriber((token) => {
            originalRequest.headers.Authorization = `Bearer ${token}`;
            resolve(axiosInstance(originalRequest));
          });
        });
      }
    } else if (
      error.response &&
      error.response.data.code === "token_not_valid" &&
      originalRequest.url === "account/auth/login/refresh"
    ) {
      this.$swal.fire({
        title: "Upss!",
        text: "Sesi login Anda telah habis. Silakan login kembali.",
        icon: "warning",
        button: "OK!",
      });
      authStore.logoutWithoutRefresh();
      window.location.href = "/auth/login";
      return Promise.reject(error);
    }

    isRefreshing = false;
    return Promise.reject(error);
  }
);

export default axiosInstance;
